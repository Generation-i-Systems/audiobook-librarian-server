539    private function normalizePath(string $path, bool $enforceBookRoot = true): string
540    {

542        if (empty(trim($path))) {
543            throw new \Exception("Path cannot be empty");
544        }
545
546        // MINOR EDGE CASE: Trim whitespace
547        $path = trim($path);
548
549        // CRITICAL SAFETY: Remove null bytes (security)
550        $path = str_replace("\0", '', $path);
551
552        // MINOR EDGE CASE: Remove control characters
553        $path = preg_replace('/[\x00-\x1F\x7F]/', '', $path);

560
561        // MINOR EDGE CASE: Remove trailing dots (Windows compatibility)
562        $path = rtrim($path, '.');

565        if ($path === '.' || $path === '..') {
566            throw new \Exception("Invalid path: {$path}");
567        }
568
569        // If already absolute, use as-is, otherwise make relative to book root
570        if (str_starts_with($path, '/')) {
571            $normalized = rtrim($path, '/');
572        } else {

574        }

581            if ($part === '' || $part === '.') {

600        $realParent = realpath($parentDir) ?: $parentDir;
601        $realNormalized = rtrim($realParent . '/' . $basename, '/');

606            return $normalizedTarget;
607        }

611            realpath($this->bookRoot) ?: null,
612            config('app.book_root'),

615        ]));
616
617        $isWithinRoot = false;
618
619        foreach ($candidateRoots as $candidateRoot) {

621
622            if ($normalizedRoot === '') {
623                continue;
624            }
625
